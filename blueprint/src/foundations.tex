\chapter{Games, order, and surreal representatives}

\section{Finite games and birthdays}

\begin{definition}[Short combinatorial game]
  \label{def:game}
  \lean{Game,Game.left,Game.right,Game.zero,Game.one}
  \leanok
  A game is a finite rooted tree written \(x=\{X^L\mid X^R\}\), where the left- and
  right-option collections \(X^L\) and \(X^R\) are finite lists of games.  The basic games are
  \[
    0=\{\,\mid\,\},\qquad 1=\{0\mid\,\}.
  \]
\end{definition}

\begin{definition}[Birthday rank]
  \label{def:birthday}
  \lean{Game.birthday}
  \leanok
  \uses{def:game}
  The rank \(\bd(x)\in\mathbb N\) is defined recursively by
  \[
    \bd\bigl(\{X^L\mid X^R\}\bigr)
      = 1+\max\bigl(\{\bd(u):u\in X^L\}\cup
                         \{\bd(v):v\in X^R\}\cup\{0\}\bigr).
  \]
  Thus this repository uses the convenient convention \(\bd(0)=1\).
\end{definition}

\begin{lemma}[Options have smaller birthday]
  \label{lem:birthday-options}
  \lean{Game.birthday_lt_of_mem_options,Game.birthday_lt_left,Game.birthday_lt_right}
  \leanok
  If \(u\in X^L\) or \(u\in X^R\), then \(\bd(u)<\bd(x)\).
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:birthday}
  Write \(x=\{X^L\mid X^R\}\), and let \(B\) be the list obtained by mapping
  \(\bd\) over \(X^L\mathbin{++}X^R\).  Membership of an option \(u\) in either
  option list first gives \(\bd(u)\in B\).  The auxiliary lemma
  \texttt{le\_maximum\_getD\_of\_mem} proves
  \[
    \bd(u)\leq B.\operatorname{maximum}.\operatorname{getD}(0).
  \]
  Its proof separates the two possible values of \texttt{List.maximum}.  The
  \texttt{none} case would force \(B\) to be empty, contrary to the displayed
  membership.  In the \texttt{some} case, the result is the usual bound by a
  finite-list maximum, implemented by \texttt{List.le\_of\_mem\_argmax}.
  Since the definition of \(\bd(x)\) is the successor of this maximum,
  \texttt{Nat.lt\_succ\_of\_le} gives \(\bd(u)<\bd(x)\).  The separate left-
  and right-option lemmas merely inject membership into
  \(X^L\mathbin{++}X^R\) and invoke this argument.
\end{proof}

\section{The recursive order}

\begin{definition}[Game comparison and equivalence]
  \label{def:game-order}
  \lean{Game.le,Game.lt,Game.eq}
  \leanok
  \uses{def:game,def:birthday}
  The relation \(x\GameLe y\) is defined recursively by
  \[
    x\GameLe y
    \quad\Longleftrightarrow\quad
    \bigl(\forall x^L\in X^L,\ \neg(y\GameLe x^L)\bigr)
    \ \wedge\
    \bigl(\forall y^R\in Y^R,\ \neg(y^R\GameLe x)\bigr).
  \]
  Strict comparison and game equivalence are
  \[
    x\GameLt y \Longleftrightarrow x\GameLe y\ \wedge\ \neg(y\GameLe x),
    \qquad
    x\GameEq y \Longleftrightarrow x\GameLe y\ \wedge\ y\GameLe x.
  \]
\end{definition}

\begin{definition}[Well-founded induction schemes]
  \label{def:game-inductions}
  \lean{Game.wf_R,Game.wf_B,Game.wf_T,Game.wf_Q}
  \leanok
  \uses{def:birthday}
  The development uses well-founded relations on one, two, three, and four games.  Their
  measures are respectively a birthday and the sums of two, three, and four birthdays.
\end{definition}

\begin{theorem}[Order calculus for games]
  \label{thm:game-order-calculus}
  \lean{Game.le_congr,Game.le_trans,Game.eq_symm,Game.eq_trans,Game.lt_trans,Game.lt_of_lt_of_le,Game.lt_of_le_of_lt}
  \leanok
  The relation \(\GameLe\) is reflexive and transitive.  Consequently \(\GameEq\) is an
  equivalence relation, \(\GameLt\) is transitive, and strict and weak inequalities compose in
  either order.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:birthday-options,def:game-order,def:game-inductions}
  For reflexivity, use well-founded induction on \(\bd(x)\) and unfold
  \(x\GameLe x\).  If \(x^L\) is a left option and one had
  \(x\GameLe x^L\), then the left-option clause of this very comparison would
  say \(\neg(x^L\GameLe x^L)\).  The induction hypothesis gives
  \(x^L\GameLe x^L\), because \(\bd(x^L)<\bd(x)\), a contradiction.  The
  right-option clause is identical with the roles reversed.  This is the Lean
  proof \texttt{Game.le\_congr}.

  For transitivity, prove the strengthened statement for a triple
  \((a,b,c)\) by induction on
  \(\bd(a)+\bd(b)+\bd(c)\).  Assume \(a\GameLe b\) and \(b\GameLe c\).
  To check the left clause of \(a\GameLe c\), let \(a^L\) be a left option and
  suppose \(c\GameLe a^L\).  The smaller triple \((b,c,a^L)\) gives
  \(b\GameLe a^L\), contradicting the left clause of \(a\GameLe b\).
  To check the right clause, let \(c^R\) be a right option and suppose
  \(c^R\GameLe a\).  Induction on the smaller triple \((c^R,a,b)\) gives
  \(c^R\GameLe b\), contradicting the right clause of \(b\GameLe c\).
  The birthday inequalities for the displayed options are exactly what proves
  that both recursive calls decrease the triple measure.

  Symmetry and transitivity of \(\GameEq\) now follow by composing its two weak
  inequalities.  For strict transitivity, compose the forward weak
  inequalities; a reverse weak inequality would, by weak transitivity,
  contradict one of the strict hypotheses.  The two mixed strict--weak laws
  are proved by the same single contradiction.
\end{proof}

\begin{theorem}[Equivalence from equivalent option lists]
  \label{thm:option-extensionality}
  \lean{Game.eq_of_equiv_options}
  \leanok
  Let \(a,b\) be games.  Suppose every left option of either game is equivalent to a left option
  of the other, and likewise every right option is equivalent to a right option of the other.
  Then \(a\GameEq b\).
\end{theorem}
\begin{proof}
  \leanok
  \uses{def:game-order,thm:game-order-calculus}
  We first prove the one-sided comparison \(a\GameLe b\).  Let \(a^L\) be a
  left option of \(a\), and suppose for contradiction that
  \(b\GameLe a^L\).  By the left-option matching hypothesis choose a left
  option \(b^L\) of \(b\) with \(a^L\GameEq b^L\).  Its forward weak
  inequality and transitivity give \(b\GameLe b^L\), contradicting the
  defining left clause of \(b\GameLe b\).  Here reflexivity supplies
  \(b\GameLe b\).

  Similarly, if \(b^R\) is a right option of \(b\) and
  \(b^R\GameLe a\), choose a matching right option \(a^R\) of \(a\).
  Transitivity gives \(a^R\GameLe a\), contradicting the right clause of
  \(a\GameLe a\).  These two contradictions establish \(a\GameLe b\).
  Applying the same one-sided argument after interchanging \(a\) and \(b\)
  establishes \(b\GameLe a\), hence \(a\GameEq b\).
\end{proof}

\section{Surreal games}

\begin{definition}[Surreal game]
  \label{def:is-surreal}
  \lean{IsSurreal,IsSurreal.isSurreal_left,IsSurreal.isSurreal_right,IsSurreal.isSurreal_zero,IsSurreal.isSurreal_one}
  \leanok
  \uses{def:game,def:game-order}
  A game \(x=\{X^L\mid X^R\}\) is surreal when every option is surreal and no right option is
  weakly below a left option:
  \[
    \forall x^L\in X^L\ \forall x^R\in X^R,
      \quad \neg(x^R\GameLe x^L).
  \]
  The games \(0\) and \(1\) are surreal, and every option of a surreal game is surreal.
\end{definition}

\begin{theorem}[A surreal game lies between its options]
  \label{thm:between-options}
  \lean{IsSurreal.left_lt,IsSurreal.lt_right,IsSurreal.xL_x_xR}
  \leanok
  If \(x\) is surreal, then
  \[
    \forall x^L\in X^L,\quad x^L\GameLt x,
    \qquad
    \forall x^R\in X^R,\quad x\GameLt x^R.
  \]
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:birthday-options,def:game-order,def:game-inductions,thm:game-order-calculus,def:is-surreal}
  We spell out the left-option case; the right-option case is its mirror.
  First prove \(x^L\GameLe x\) by well-founded induction on \(\bd(x)\).
  For a left option \(z\) of \(x^L\), suppose \(x\GameLe z\).  The induction
  hypothesis, applied inside the surreal option \(x^L\), gives
  \(z\GameLe x^L\).  But the left-option clause of \(x\GameLe z\) together
  with \(x^L\in X^L\) forbids precisely this comparison.  For a right option
  \(x^R\) of \(x\), the required noncomparison
  \(\neg(x^R\GameLe x^L)\) is exactly the separation field in the proof that
  \(x\) is surreal.  Thus both recursive clauses of \(x^L\GameLe x\) hold.

  The comparison is strict because the reverse inequality
  \(x\GameLe x^L\) would use its left-option clause to assert
  \(\neg(x^L\GameLe x^L)\), contradicting reflexivity.  Dually, induction
  proves \(x\GameLe x^R\), and the right-option clause rules out
  \(x^R\GameLe x\).  Combining the two statements gives
  \(x^L\GameLt x\GameLt x^R\).
\end{proof}

\begin{theorem}[Totality and trichotomy for surreal games]
  \label{thm:surreal-totality}
  \lean{IsSurreal.le_of_not_le,IsSurreal.totality,IsSurreal.trichotomy}
  \leanok
  If \(x\) and \(y\) are surreal, then
  \[
    x\GameLe y\ \vee\ y\GameLe x,
  \]
  and one of the three alternatives \(x\GameLt y\), \(x\GameEq y\), or
  \(y\GameLt x\) holds.
\end{theorem}
\begin{proof}
  \leanok
  \uses{def:game-order,thm:game-order-calculus,thm:between-options}
  Assume \(\neg(x\GameLe y)\).  Unfolding \(\GameLe\), pushing the negation
  through its conjunction and universal quantifiers, gives one of two
  witnesses.
  Either there is a left option \(x^L\) with \(y\GameLe x^L\), or there is a
  right option \(y^R\) with \(y^R\GameLe x\).  In the first case the
  between-options theorem gives \(x^L\GameLe x\), so transitivity yields
  \(y\GameLe x\).  In the second case it gives
  \(y\GameLe y^R\), and transitivity again yields \(y\GameLe x\).  Thus
  failure of \(x\GameLe y\) always implies the reverse weak comparison.

  Totality follows by deciding \(x\GameLe y\).  To obtain trichotomy, first
  use totality and then decide the reverse comparison: two weak comparisons
  give \(x\GameEq y\); exactly one gives the corresponding strict comparison.
\end{proof}

\begin{definition}[Surreal representatives]
  \label{def:surreal-representative}
  \lean{Surreal,Surreal.sr_zero,Surreal.sr_one,Surreal.leftOption,Surreal.rightOption}
  \leanok
  \uses{def:is-surreal}
  The type \texttt{Surreal} consists of a game together with a proof that it is surreal.  It is
  a type of representatives; equivalent game trees are not yet identified.
\end{definition}
