\chapter{Addition, negation, and the additive quotient}

\section{Operations on games}

\begin{definition}[Recursive sum]
  \label{def:game-add}
  \lean{Game.add,add_left,add_right,mem_add_left_iff,mem_add_right_iff}
  \leanok
  \uses{def:game,def:birthday}
  For games \(x=\{X^L\mid X^R\}\) and \(y=\{Y^L\mid Y^R\}\), define
  \[
    x+y=
    \bigl\{\,x^L+y,\ x+y^L\ \bigm|\ x^R+y,\ x+y^R\,\bigr\},
  \]
  with one option for every member of the indicated finite lists.
\end{definition}

\begin{theorem}[Basic addition laws]
  \label{thm:game-add-laws}
  \lean{Game.add_zero',Game.zero_add',Game.add_zero,Game.zero_add,Game.add_comm,Game.add_assoc}
  \leanok
  For all games \(a,b,c\),
  \[
    a+0=a,\qquad 0+a=a,\qquad a+b\GameEq b+a,
    \qquad (a+b)+c=a+(b+c).
  \]
  The zero and associativity identities hold as literal equalities of game trees; commutativity
  is asserted up to game equivalence.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:birthday-options,def:game-inductions,thm:game-order-calculus,thm:option-extensionality,def:game-add}
  For \(a+0=a\), perform well-founded induction on \(\bd(a)\), destruct
  \(a=\{A^L\mid A^R\}\), and simplify the recursive definition.  The left
  list of \(a+0\) is the map \(u\mapsto u+0\) over \(A^L\), and similarly on
  the right.  The induction hypotheses identify every mapped entry literally
  with \(u\), so congruence of list maps proves equality of the two game
  trees.  The proof of \(0+a=a\) is identical.

  For commutativity, induct on \(\bd(a)+\bd(b)\).  A left option of \(a+b\)
  is either \(a^L+b\) or \(a+b^L\).  The smaller-pair induction hypothesis
  makes it equivalent respectively to \(b+a^L\) or \(b^L+a\), which is a
  left option of \(b+a\).  The two right-option cases are the same, and the
  converse matching is obtained by swapping \(a,b\).  Option extensionality
  then gives \(a+b\GameEq b+a\).

  Associativity is stronger: it is literal equality of the two game trees.
  The proof first inducts well foundedly on \(a\), while generalizing \(b,c\);
  inside that induction it inducts on \(b\), while generalizing \(c\); finally
  it inducts on \(c\).  Consequently the available induction hypotheses are
  \[
    \begin{array}{ll}
      ((a^L+b)+c)=a^L+(b+c) & (a^L\in A^L),\\
      ((a+b^L)+c)=a+(b^L+c) & (b^L\in B^L),\\
      ((a+b)+c^L)=a+(b+c^L) & (c^L\in C^L),
    \end{array}
  \]
  and the analogous three identities for right options.  Each recursive call
  is legitimate because the option in the indicated coordinate has smaller
  birthday; generalizing the later coordinates is what permits the unchanged
  games in the other two positions.

  Apply \texttt{Game.ext}.  For the left lists, expanding the definition of
  addition twice gives
  \[
    \begin{aligned}
      ((a+b)+c)^L
       ={}&[\,(a^L+b)+c:a^L\in A^L\,]\\
          &{}\mathbin{++}[\,(a+b^L)+c:b^L\in B^L\,]\\
          &{}\mathbin{++}[\,(a+b)+c^L:c^L\in C^L\,],\\[2mm]
      (a+(b+c))^L
       ={}&[\,a^L+(b+c):a^L\in A^L\,]\\
          &{}\mathbin{++}[\,a+(b^L+c):b^L\in B^L\,]\\
          &{}\mathbin{++}[\,a+(b+c^L):c^L\in C^L\,].
    \end{aligned}
  \]
  Thus the lists already contain the same three families in the same order;
  only their entries differ.  The first, second, and third induction
  hypotheses above give pointwise equality on \(A^L,B^L,C^L\), respectively.
  The helper \texttt{list\_map\_congr} promotes each pointwise identity to an
  equality of mapped lists, and \texttt{append\_map3\_congr} combines the
  three equalities.  The simplifications \texttt{map\_append},
  \texttt{map\_map}, and \texttt{append\_assoc} merely expose these displayed
  normal forms.  Replacing \(A^L,B^L,C^L\) by \(A^R,B^R,C^R\) proves equality
  of the right lists by the same three induction hypotheses.  Hence
  \texttt{Game.ext} yields \(((a+b)+c)=a+(b+c)\) as a literal equality.
\end{proof}
\begin{theorem}[Translation and monotonicity]
  \label{thm:game-add-order}
  \lean{Game.add_le_add_right,Game.add_right_cancel,Game.add_le_add,Game.add_equal,Game.add_lt_le,Game.add_le_lt,Game.add_lt_left_left,Game.add_lt_right_right}
  \leanok
  Addition preserves and reflects weak order in either variable.  In particular,
  \[
    a\GameLe b\quad\Longleftrightarrow\quad a+c\GameLe b+c,
  \]
  and
  \[
    a\GameLe c,\ b\GameLe d\Longrightarrow a+b\GameLe c+d.
  \]
  If either input comparison is strict and the other weak, the resulting comparison is strict.
  Addition also respects \(\GameEq\).
\end{theorem}
\begin{proof}
  \leanok
  \uses{def:game-order,def:game-inductions,thm:game-order-calculus,def:game-add,thm:game-add-laws}
  The central lemma proves both directions of
  \(a\GameLe b\Longleftrightarrow a+c\GameLe b+c\) simultaneously by
  induction on \(\bd(a)+\bd(b)+\bd(c)\).

  For reflection, suppose \(a+c\GameLe b+c\).  If \(a^L\) is a left option
  and \(b\GameLe a^L\), the induction hypothesis translates this to
  \(b+c\GameLe a^L+c\).  Transitivity with the assumed comparison contradicts
  the fact that \(a^L+c\) is a left option of \(a+c\).  A right option
  \(b^R\) is treated dually, using \(b^R+c\) as a right option of \(b+c\).

  For preservation, unfold both sums.  A left option of \(a+c\) is either
  \(a^L+c\) or \(a+c^L\), and a right option of \(b+c\) is either
  \(b^R+c\) or \(b+c^R\).  In the first-coordinate cases the induction
  hypothesis translates a hypothetical forbidden comparison back to
  \(b\GameLe a^L\) or \(b^R\GameLe a\), contradicting \(a\GameLe b\).
  In the common-coordinate cases, recursive cancellation translates it to a
  comparison of \(c\) with one of its own options, which contradicts the
  corresponding clause of reflexivity \(c\GameLe c\).

  To prove two-variable monotonicity, first change the first summand and then
  the second; commutativity supplies the left-variable form, and transitivity
  composes the two changes.  If one input comparison is strict, weak
  monotonicity gives the forward comparison.  A reverse comparison of the
  sums would cancel the unchanged summand and contradict the strict input.
  Applying these arguments in both directions also proves compatibility with
  \(\GameEq\).
\end{proof}

\begin{definition}[Negation]
  \label{def:game-neg}
  \lean{Game.neg,neg_left_def,neg_right_def,Game.neg_zero}
  \leanok
  \uses{def:game,def:birthday}
  Negation interchanges the two option lists and recursively negates every option:
  \[
    -\{X^L\mid X^R\}=\{-X^R\mid -X^L\}.
  \]
\end{definition}

\begin{theorem}[Negation reverses order]
  \label{thm:game-neg-order}
  \lean{Game.neg_le_neg,Game.neg_congr,Game.neg_lt_neg}
  \leanok
  For all games \(a,b\),
  \[
    a\GameLe b\Longleftrightarrow -b\GameLe-a,
    \qquad
    a\GameLt b\Longleftrightarrow -b\GameLt-a,
  \]
  and game equivalence is preserved by negation.
\end{theorem}
\begin{proof}
  \leanok
  \uses{def:game-order,def:game-inductions,def:game-neg,thm:game-order-calculus}
  Prove the weak-order equivalence in both directions by induction on
  \(\bd(a)+\bd(b)\).  A left option of \(-b\) has the form \(-b^R\), and a
  right option of \(-a\) has the form \(-a^L\).  Thus, in the forward
  direction, a hypothetical comparison
  \(-a\GameLe -b^R\) is converted by the smaller-pair induction hypothesis to
  \(b^R\GameLe a\), contradicting the right clause of \(a\GameLe b\).
  Likewise, \(-a^L\GameLe-b\) converts to \(b\GameLe a^L\), contradicting
  the left clause.  These are exactly the two clauses required for
  \(-b\GameLe-a\).

  Conversely, start from \(-b\GameLe-a\) and apply the same two recursive
  equivalences to the option clauses for \(a\GameLe b\).  All recursive calls
  decrease because an option has smaller birthday.  Applying the weak-order
  equivalence to both components of \(\GameEq\) proves preservation of game
  equivalence; applying it to the weak inequality and its negated reverse
  proves the strict-order equivalence.
\end{proof}

\begin{theorem}[Additive inverses]
  \label{thm:game-add-inverse}
  \lean{Game.add_neg,Game.neg_add}
  \leanok
  Every game satisfies
  \[
    x+(-x)\GameEq0,
    \qquad
    (-x)+x\GameEq0.
  \]
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:birthday-options,thm:game-order-calculus,def:game-add,thm:game-add-laws,def:game-neg}
  Induct on \(\bd(x)\) and prove both inequalities defining
  \(x+(-x)\GameEq0\).  Since \(0\) has no options, one half of each unfolded
  order statement is vacuous.  A left option of \(x+(-x)\) is either
  \(x^L+(-x)\) or \(x+(-x^R)\).  In the first case, the corresponding right
  option \(x^L+(-x^L)\) of \(x^L+(-x)\) is equivalent to zero by induction;
  in the second, use the corresponding option \(x^R+(-x^R)\).  Therefore a
  hypothetical comparison of zero with the given left option violates its
  right-option clause.

  Dually, every right option of \(x+(-x)\) contains an appropriate smaller
  self-cancelling game as a left option.  The induction hypothesis then rules
  out comparison of that right option with zero.  Hence
  \(x+(-x)\GameEq0\).  Commutativity followed by this result gives
  \((-x)+x\GameEq0\).
\end{proof}

\section{Closure and quotients}

\begin{theorem}[Surreal games are closed under addition and negation]
  \label{thm:surreal-add-closure}
  \lean{Surreal.add_isSurreal,Surreal.neg_isSurreal,Surreal.add,Surreal.neg}
  \leanok
  If \(a,b\) are surreal representatives, then the games \(a+b\) and \(-a\) are surreal.
  Hence addition and negation define operations on the representative type \texttt{Surreal}.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:game-order-calculus,thm:between-options,thm:surreal-totality,def:surreal-representative,def:game-add,thm:game-add-order,def:game-neg,thm:game-neg-order}
  For addition, use well-founded induction on the pair of birthdays.  Each
  option of \(a+b\) changes exactly one coordinate to one of its options, so
  it is surreal by the induction hypothesis.  For separation, choose a left
  and a right option and split each membership according to its source.  This
  gives four cases.  If both options change the first coordinate, use
  \(a^L\GameLt a\GameLt a^R\) and strict additive monotonicity; if both change
  the second, use the analogous inequalities for \(b\).  In either mixed
  case, one coordinate comparison is strict and the other weak, so the two
  mixed monotonicity lemmas again show that the chosen left option is strictly
  below the chosen right option.  Strictness is exactly the required
  noncomparison across the cut.

  For negation, induct on the representative's birthday.  Its left options
  are \(-a^R\) and its right options are \(-a^L\).  Original surreal
  separation says \(\neg(a^R\GameLe a^L)\); order reversal turns a
  hypothetical \(-a^L\GameLe-a^R\) back into this forbidden comparison.
  Each negated option is surreal by the smaller-birthday induction hypothesis.
  These separation and closure fields construct the required
  \texttt{IsSurreal} values, after which the operations on representatives are
  obtained by pairing the game operations with those proofs.
\end{proof}

\begin{definition}[Auxiliary quotient of all games]
  \label{def:game-quotient}
  \lean{Game.GameQ,Game.q,Game.instAddCommGroupGameQ,Game.instPartialOrderGameQ,Game.instIsOrderedAddMonoidGameQ}
  \leanok
  \uses{def:game-order,def:game-add,def:game-neg}
  Let \(\mathcal G/{\GameEq}\) be the quotient of all finite games by game equivalence.  The
  repository equips this quotient with a partially ordered additive commutative group
  structure.  This auxiliary quotient is used for algebraic rearrangement in later product
  proofs; totality is not asserted for arbitrary games.
\end{definition}
\begin{proof}
  \leanok
  \uses{thm:game-order-calculus,thm:game-add-laws,thm:game-add-order,thm:game-neg-order,thm:game-add-inverse}
  Game equivalence is made into a \texttt{Setoid}, and
  \texttt{Quotient.map\(_2\)} defines negation and addition.  Their
  well-definedness obligations are exactly \texttt{Game.neg\_congr} and
  \texttt{Game.add\_equal}.  Each additive group law is then proved by
  quotient induction: zero and associativity use the literal game equalities,
  while commutativity and inverse use \texttt{Quotient.sound} with the
  corresponding game equivalences.

  The order is defined by \texttt{Quotient.lift\(_2\)}.  If representatives
  are replaced by equivalent games, the two components of those equivalences,
  composed with transitivity, prove that the truth of \(a\GameLe b\) is
  unchanged.  Reflexivity and transitivity therefore descend directly;
  antisymmetry follows because inequalities in both directions are precisely
  game equivalence, hence equality in the quotient.  Finally, translation
  monotonicity descends from \texttt{Game.add\_le\_add\_right}, with
  commutativity used to place the common summand in the convention required by
  the ordered-additive-monoid instance.
\end{proof}

\begin{definition}[Short surreal numbers]
  \label{def:surreal-number}
  \lean{Surreal.SurrealNumber,Surreal.SurrealNumber.add,Surreal.SurrealNumber.neg}
  \leanok
  \uses{def:surreal-representative,thm:surreal-add-closure}
  The type \(\NoShort\), called \texttt{SurrealNumber} in Lean, is the quotient of surreal
  representatives by \(a\GameEq b\).  Addition, zero, one, and negation descend to this quotient.
\end{definition}

\begin{theorem}[Linear ordered additive commutative group]
  \label{thm:ordered-add-group}
  \lean{Surreal.instLinearOrderSurrealNumber,Surreal.instAddCommGroupSurrealNumber,Surreal.instIsOrderedAddMonoidSurrealNumber}
  \leanok
  The quotient \(\NoShort\) is a linearly ordered additive commutative group, and addition is
  compatible with its order.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:surreal-totality,thm:game-add-laws,thm:game-add-order,thm:game-add-inverse,def:surreal-number}
  The construction repeats the preceding quotient argument on the subtype of
  surreal representatives.  For order well-definedness, replace either
  representative by an equivalent one and compose the appropriate weak
  inequalities.  Quotient induction then supplies reflexivity and
  transitivity; antisymmetry is \texttt{Quotient.sound}, and totality is the
  game-level totality theorem applied to the representatives' surrealness
  proofs.  This produces the linear order.

  The additive group laws are again checked by quotient induction.  Literal
  game equalities are promoted through \texttt{Subtype.ext}; laws stated only
  up to \(\GameEq\) are promoted through \texttt{Quotient.sound}.  In
  particular, the inverse law uses \texttt{Game.add\_neg}.  Finally, if
  \(a\GameLe b\), two-variable game monotonicity applied to this comparison
  and reflexivity of the common summand gives \(c+a\GameLe c+b\), which is
  the compatibility axiom for the quotient order.
\end{proof}
