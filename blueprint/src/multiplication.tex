\section{The recursive product}

\begin{definition}[Product and product-option expression]
  \label{def:game-mul}
  \lean{Game.mul,Game.mulOpt4,mem_mul_left,mem_mul_right}
  \leanok
  \uses{def:game,def:birthday,def:game-add,def:game-neg}
  Put
  \[
    \mulopt(x',y;x,y')=x'y+xy'-x'y'.
  \]
  For \(x=\{X^L\mid X^R\}\) and \(y=\{Y^L\mid Y^R\}\), multiplication is the recursive cut
  \[
    xy=
    \left\{
      \begin{array}{ll}
        \mulopt(x^L,y;x,y^L),&\mulopt(x^R,y;x,y^R)
      \end{array}
      \ \middle|\
      \begin{array}{ll}
        \mulopt(x^L,y;x,y^R),&\mulopt(x^R,y;x,y^L)
      \end{array}
    \right\},
  \]
  with all choices of the displayed options.  The birthday-sum measure makes every recursive
  call smaller.
\end{definition}

\begin{theorem}[Zero, one, and commutativity for game multiplication]
  \label{thm:game-mul-laws}
  \lean{Game.mul_zero,Game.zero_mul,Game.mul_comm,Game.mul_one,Game.one_mul}
  \leanok
  For all games \(a,b\),
  \[
    a0\GameEq0,\qquad 0a\GameEq0,\qquad ab\GameEq ba,
    \qquad a1\GameEq a,\qquad 1a\GameEq a.
  \]
  The repository also proves literal game-tree equalities for multiplication on the right by
  zero and one.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:birthday-options,def:game-inductions,thm:option-extensionality,thm:game-add-laws,thm:game-add-inverse,def:game-mul}
  In \(a0\), the second factor has no options.  Expanding the product therefore
  leaves only flattened maps over empty lists, so simplification proves the
  literal equality \(a0=0\); \(0a\GameEq0\) then follows by commutativity (or
  by the symmetric simplification).

  Commutativity is proved by well-founded induction on
  \(\bd(a)+\bd(b)\).  Split an option according to the four constructors in
  the product definition.  For a same-side example, the \(LL\) option of
  \(ab\) determined by \((a^L,b^L)\) is
  \[
    \mulopt(a^L,b;a,b^L)=a^Lb+ab^L-a^Lb^L.
  \]
  The \(LL\) option of \(ba\) determined by \((b^L,a^L)\) is
  \[
    \mulopt(b^L,a;b,a^L)=b^La+ba^L-b^La^L.
  \]
  The induction hypotheses for the three smaller pairs
  \((a^L,b)\), \((a,b^L)\), and \((a^L,b^L)\) identify, respectively,
  \(a^Lb\GameEq ba^L\), \(ab^L\GameEq b^La\), and
  \(a^Lb^L\GameEq b^La^L\).  Additive commutativity, addition congruence,
  and negation congruence therefore make the two displayed options
  equivalent.  The \(RR\) case is the identical calculation with
  \(a^L,b^L\) replaced by \(a^R,b^R\):
  \[
    \mulopt(a^R,b;a,b^R)\GameEq\mulopt(b^R,a;b,a^R).
  \]

  A mixed-side option changes family when the factors are swapped.  The
  \(LR\) option
  \[
    \mulopt(a^L,b;a,b^R)=a^Lb+ab^R-a^Lb^R
  \]
  is matched not with another \(LR\) option, but with the \(RL\) option
  \[
    \mulopt(b^R,a;b,a^L)=b^Ra+ba^L-b^Ra^L
  \]
  of \(ba\).  The induction hypotheses for \((a^L,b)\), \((a,b^R)\), and
  \((a^L,b^R)\) identify its three terms after the same additive
  rearrangement.  Dually,
  \[
    \mulopt(a^R,b;a,b^L)\GameEq\mulopt(b^L,a;b,a^R),
  \]
  so an \(RL\) option of \(ab\) matches an \(LR\) option of \(ba\).
  Repeating these matches from \(ba\) back to \(ab\), for both the left and
  right option lists, supplies the four hypotheses of option extensionality
  and proves \(ab\GameEq ba\).

  Finally, prove the literal equality \(a1=a\) by induction on \(\bd(a)\).
  Because \(1=\{0\mid\}\), expansion reduces the product's left and right
  lists to maps of the smaller products \(a'1\); all additional zero-product
  and additive-inverse terms simplify.  The induction hypotheses identify
  these lists with the option lists of \(a\).  The law \(1a\GameEq a\) follows
  by product commutativity and \(a1=a\).
\end{proof}
\begin{theorem}[Distributivity on games]
  \label{thm:game-distrib}
  \lean{Game.mul_distrib_tri,Game.mul_distrib,Game.mul_distrib_right}
  \leanok
  For all games \(a,b,c\),
  \[
    a(b+c)\GameEq ab+ac,
    \qquad
    (a+b)c\GameEq ac+bc.
  \]
\end{theorem}
\begin{proof}
  \leanok
  \uses{def:game-inductions,thm:option-extensionality,thm:game-add-laws,thm:game-add-order,def:game-quotient,def:game-mul,thm:game-mul-laws}
  Prove \(a(b+c)\GameEq ab+ac\) by well-founded induction on
  \(\bd(a)+\bd(b)+\bd(c)\).  Four matching lemmas treat, in each direction,
  left and right options.  For an option of \(a(b+c)\), first split whether
  it comes from an option of \(a\) or an option of \(b+c\), and in the latter
  case split once more according to whether the option came from \(b\) or
  \(c\).  The matching option of \(ab+ac\) changes the same coordinate in the
  corresponding summand.  Conversely, an option of the sum changes either
  \(ab\) or \(ac\), and expansion of that product determines the matching
  option on the left.

  After these membership splits, the remaining assertion is an equivalence
  between two expanded \(\mulopt\) expressions.  Map the games to
  \(\mathcal G/{\GameEq}\).  The three relevant smaller distributivity
  hypotheses replace the recursively occurring products, and congruence of
  \(\mulopt\) transports those replacements.  In the additive quotient,
  \texttt{abel} normalizes the sums and negatives and proves equality; pulling
  this equality back gives game equivalence.  Option extensionality applied to
  the four matching lemmas completes left distributivity.  For right
  distributivity, commute the outer product, apply the established left law,
  and commute the two resulting products.
\end{proof}

\begin{definition}[Dershowitz--Manna goal order]
  \label{def:dm-goals}
  \lean{Goal,μA,μB,μGoal,GoalLT,wfGoal}
  \leanok
  \uses{def:birthday}
  Three kinds of induction goal are used:
  \[
    A(x,y),\qquad B(x_1,x_2,y),\qquad C(x_1,x_2,y).
  \]
  Their measures are the multisets
  \[
    \mu_A(x,y)=\{\bd(x),\bd(y)\},\qquad
    \mu_B(x_1,x_2,y)=\mu_C(x_1,x_2,y)
      =\{\bd(x_1),\bd(x_2),\bd(y)\}.
  \]
  Goals are ordered by the Dershowitz--Manna multiset extension of \(<\) on
  \(\mathbb N\).  Concretely, \(U<_{\rm DM}V\) when there are multisets
  \(X,Y,Z\), with \(Y\ne\varnothing\), such that
  \[
    V=X+Y,\qquad U=X+Z,
  \]
  and every element of \(Z\) is strictly smaller than some element removed in
  \(Y\).  Thus the common part \(X\) is left untouched, while the nonempty
  part \(Y\) is replaced by smaller data \(Z\).

  Three consequences of this definition are used repeatedly.  First, one
  birthday may be replaced by the smaller birthday of an option while all
  other birthdays remain in \(X\).  Second, one birthday may be replaced by
  two smaller option birthdays; this is what makes a three-variable \(B\)- or
  \(C\)-goal smaller than a two-variable \(A\)-goal.  Conversely, a birthday
  may simply be deleted by taking \(Z=\varnothing\), so the appropriate
  two-variable \(A\)-goals are smaller than a \(B\)- or \(C\)-goal.  Third,
  transitivity composes successive coordinate replacements, and the multiset
  measure is insensitive to permuting the coordinates.  The lemmas
  \texttt{goal\_lt\_A\(_1\)}, \texttt{goal\_lt\_B\(_i\)},
  \texttt{goal\_lt\_C\(_i\)}, and the \texttt{from\_A}/\texttt{from\_B}
  variants in \texttt{Order.lean} package exactly these moves.  Since
  \(<\) on \(\mathbb N\) is well founded, its Dershowitz--Manna extension is
  well founded; pulling it back along \(\mu_{\rm Goal}\) gives
  \texttt{wfGoal}.
\end{definition}

\begin{definition}[Cross inequalities]
  \label{def:cross-conditions}
  \lean{CLeft,CRight,MulCrossLe,MulCrossLt}
  \leanok
  \uses{def:game-order,def:game-add,def:game-mul}
  For \(x_1\GameLt x_2\), the two forms needed to compare product options are
  \[
    \begin{aligned}
      C_L(x_1,x_2;y,y^L)&:\quad
        x_1y+x_2y^L\GameLt x_1y^L+x_2y,\\
      C_R(x_1,x_2;y,y^R)&:\quad
        x_1y^R+x_2y\GameLt x_1y+x_2y^R.
    \end{aligned}
  \]
  The file \texttt{MulOpt.lean} also packages weak and strict four-product cross inequalities
  so that endpoints can be transported or replaced algebraically.
\end{definition}

\begin{lemma}[Terminal product-option bounds]
  \label{lem:mulopt-bounds}
  \lean{mulOpt4_LL_lt_product,mulOpt4_RR_lt_product,product_lt_mulOpt4_LR,product_lt_mulOpt4_RL}
  \leanok
  Assuming the appropriate multiplicative congruence and one of the conditions \(C_L,C_R\),
  each left product option is strictly below the target product and each right product option
  is strictly above it.  There are four lemmas, corresponding to the \(LL,RR,LR,RL\) option
  configurations in the definition of multiplication.
\end{lemma}
\begin{proof}
  \leanok
  \uses{thm:game-add-order,def:game-quotient,def:cross-conditions}
  All four proofs take place in the ordered additive group
  \(\mathcal G/{\GameEq}\); write \(q(g)\) for the class of a game \(g\).
  This permits cancellation and additive normalization even though the
  corresponding game trees are not literally equal.

  For the \(LL\) bound, assume
  \[
    ay^L\GameEq by^L
    \quad\text{and}\quad
    C_L(a^L,b;y,y^L):
    a^Ly+by^L\GameLt a^Ly^L+by.
  \]
  Map the strict inequality to the quotient and add
  \(-q(a^Ly^L)\) to both sides.  Replace \(q(by^L)\) by \(q(ay^L)\) using
  the equivalence hypothesis.  The translated inequality is
  \[
    q(a^Ly)+q(ay^L)-q(a^Ly^L)<q(by),
  \]
  which is exactly
  \(q(\mulopt(a^L,y;a,y^L))<q(by)\).  Reflection of quotient strict order
  gives
  \[
    \mulopt(a^L,y;a,y^L)\GameLt by.
  \]

  For \(RR\), assume \(ay^R\GameEq by^R\) and
  \[
    C_R(b,a^R;y,y^R):
    by^R+a^Ry\GameLt by+a^Ry^R.
  \]
  Add \(-q(a^Ry^R)\), replace \(q(by^R)\) by \(q(ay^R)\), and commute the
  two positive summands.  This yields
  \[
    q(\mulopt(a^R,y;a,y^R))
      =q(a^Ry)+q(ay^R)-q(a^Ry^R)<q(by),
  \]
  hence \(\mulopt(a^R,y;a,y^R)\GameLt by\).

  The mixed cases give upper bounds for right product options.  For \(LR\),
  assume \(by^R\GameEq ay^R\) and
  \[
    C_R(b^L,a;y,y^R):
    b^Ly^R+ay\GameLt b^Ly+ay^R.
  \]
  Adding \(-q(b^Ly^R)\) and replacing \(q(ay^R)\) by \(q(by^R)\) gives
  \[
    q(ay)<q(b^Ly)+q(by^R)-q(b^Ly^R)
      =q(\mulopt(b^L,y;b,y^R)),
  \]
  so \(ay\GameLt\mulopt(b^L,y;b,y^R)\).

  Finally, for \(RL\), assume \(by^L\GameEq ay^L\) and
  \[
    C_L(a,b^R;y,y^L):
    ay+b^Ry^L\GameLt ay^L+b^Ry.
  \]
  Add \(-q(b^Ry^L)\), replace \(q(ay^L)\) by \(q(by^L)\), and commute the
  remaining positive terms.  The result is
  \[
    q(ay)<q(b^Ry)+q(by^L)-q(b^Ry^L)
      =q(\mulopt(b^R,y;b,y^L)),
  \]
  hence \(ay\GameLt\mulopt(b^R,y;b,y^L)\).  These are precisely the two
  lower bounds for \(LL,RR\) left options and the two upper bounds for
  \(LR,RL\) right options used in the \(B\)-case of the Conway induction.
\end{proof}

\begin{theorem}[Simultaneous Conway theorem]
  \label{thm:abc-main}
  \lean{Conway.Holds,Conway.ABC_main}
  \leanok
  The following assertions hold simultaneously for every well-formed input:
  \begin{enumerate}
    \item \(A(x,y)\): if \(x,y\) are surreal, then \(xy\) is surreal;
    \item \(B(x_1,x_2,y)\): if \(x_1\GameEq x_2\) and all three games are surreal, then
      \(x_1y\GameEq x_2y\);
    \item \(C(x_1,x_2,y)\): if \(x_1\GameLt x_2\) and all three games are surreal, then
      \(C_L(x_1,x_2;y,y^L)\) holds for every \(y^L\), and
      \(C_R(x_1,x_2;y,y^R)\) holds for every \(y^R\).
  \end{enumerate}
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:game-order-calculus,thm:option-extensionality,thm:between-options,thm:surreal-totality,thm:game-add-order,thm:game-mul-laws,def:dm-goals,def:cross-conditions,lem:mulopt-bounds}
  The Lean proof performs one induction on \texttt{wfGoal}; its induction
  hypothesis is available for every goal smaller in the Dershowitz--Manna
  order.  The multiset lemmas in \texttt{Order.lean} justify not only replacing
  one game by an option, but also replacing a two-birthday \(A\)-goal by a
  three-birthday \(B\)- or \(C\)-goal when the extra birthdays are strictly
  smaller than the one removed.  We treat the three constructors separately.

  In case \(A(x,y)\), first prove that every product option is surreal.  For
  example, if \(x'\) and \(y'\) are options of \(x\) and \(y\), respectively,
  then
  \[
    \mulopt(x',y;x,y')=x'y+xy'-x'y'.
  \]
  The lower goals \(A(x',y)\), \(A(x,y')\), and \(A(x',y')\) prove the three
  displayed products surreal.  The first two are lower by replacing one
  birthday, and the third by replacing both; addition and negation closure
  then prove the whole \(\mulopt\) expression surreal.  This argument applies
  to the \(LL,RR,LR,RL\) option families.

  We now make the separation argument explicit.  Put
  \(P(u,v)=\mulopt(u,y;x,v)\).  A left option of \(xy\) is
  \[
    P(x^L,y^L)\quad(LL),
    \qquad P(x^R,y^R)\quad(RR),
  \]
  whereas a right option is
  \[
    P(x^L,y^R)\quad(LR),
    \qquad P(x^R,y^L)\quad(RL).
  \]
  Consider first an \(LL\)-versus-\(LR\) pair
  \(P(x^L_1,y^L)\) and \(P(x^L_2,y^R)\).  Trichotomy compares
  \(x^L_1,x^L_2\), while surreal separation supplies
  \(y^L\GameLt y^R\).
  \begin{itemize}
    \item If \(x^L_1\GameLt x^L_2\), lower \(C\)-goals give
      \[
        P(x^L_1,y^L)\GameLt P(x^L_2,y^L)
          \GameLt P(x^L_2,y^R).
      \]
      The first step moves the \(x\)-slot; the second moves the \(y\)-slot.
    \item If \(x^L_1\GameEq x^L_2\), first move the \(y\)-slot and then use
      a lower \(B\)-goal in the \(x\)-slot:
      \[
        P(x^L_1,y^L)\GameLt P(x^L_1,y^R)
          \GameLe P(x^L_2,y^R).
      \]
    \item If \(x^L_2\GameLt x^L_1\), move \(y^L\) to \(y^R\) first.  At a
      right \(y\)-option the \(x\)-slot movement has the opposite orientation,
      giving
      \[
        P(x^L_1,y^L)\GameLt P(x^L_1,y^R)
          \GameLt P(x^L_2,y^R).
      \]
  \end{itemize}

  For an \(LL\)-versus-\(RL\) pair, write the options as
  \(P(x^L,y^L_1)\) and \(P(x^R,y^L_2)\).  Here
  \(x^L\GameLt x^R\) is fixed, and trichotomy is applied to
  \(y^L_1,y^L_2\).  Its three branches produce, respectively,
  \[
    \begin{aligned}
      P(x^L,y^L_1)&\GameLt P(x^L,y^L_2)
                       \GameLt P(x^R,y^L_2),\\
      P(x^L,y^L_1)&\GameLt P(x^R,y^L_1)
                       \GameLe P(x^R,y^L_2),\\
      P(x^L,y^L_1)&\GameLt P(x^R,y^L_1)
                       \GameLt P(x^R,y^L_2).
    \end{aligned}
  \]
  The middle branch uses a lower \(B\)-goal; the other steps use lower
  \(C\)-goals.  The \(RR\)-versus-\(LR\) and
  \(RR\)-versus-\(RL\) arguments are the right-option mirrors: the former
  compares two \(y^R\)'s and uses \(x^L\GameLt x^R\), while the latter
  compares two \(x^R\)'s and uses \(y^L\GameLt y^R\).  These four strict
  separation results, together with the option-surrealness proof above, fill
  the two fields of \texttt{IsSurreal} for \(xy\).

  In case \(B(a,b,y)\), assume \(a\GameEq b\) and prove
  \(ay\GameLe by\).  Unfolding the game order produces two universal clauses.
  The left clause says that for every left option \(L\) of \(ay\), one must
  prove \(\neg(by\GameLe L)\).  Splitting membership gives two cases.
  \begin{itemize}
    \item For \(L=\mulopt(a^L,y;a,y^L)\), a lower \(B\)-goal gives
      \(ay^L\GameEq by^L\).  Since
      \(a^L\GameLt a\GameLe b\), a lower \(C\)-goal gives
      \(C_L(a^L,b;y,y^L)\).  The \(LL\) terminal bound therefore gives
      \(L\GameLt by\), whose strictness contains
      \(\neg(by\GameLe L)\).
    \item For \(L=\mulopt(a^R,y;a,y^R)\), the lower \(B\)-goal gives
      \(ay^R\GameEq by^R\).  From
      \(b\GameLe a\GameLt a^R\), a lower \(C\)-goal gives
      \(C_R(b,a^R;y,y^R)\).  The \(RR\) terminal bound again gives
      \(L\GameLt by\).
  \end{itemize}

  The right clause of \(ay\GameLe by\) says that every right option \(R\) of
  \(by\) must satisfy \(\neg(R\GameLe ay)\).  Again there are two cases.
  \begin{itemize}
    \item If \(R=\mulopt(b^L,y;b,y^R)\), the symmetric lower \(B\)-goal gives
      \(by^R\GameEq ay^R\), and
      \(b^L\GameLt b\GameLe a\) gives
      \(C_R(b^L,a;y,y^R)\).  The \(LR\) terminal bound yields
      \(ay\GameLt R\).
    \item If \(R=\mulopt(b^R,y;b,y^L)\), use
      \(by^L\GameEq ay^L\) and
      \(a\GameLe b\GameLt b^R\) to obtain
      \(C_L(a,b^R;y,y^L)\).  The \(RL\) bound again yields
      \(ay\GameLt R\).
  \end{itemize}
  Thus both clauses of \(ay\GameLe by\) hold.  To obtain \(by\GameLe ay\),
  apply this same one-sided argument with \(a,b\) interchanged and use
  symmetry of \(a\GameEq b\).  This is not a recursive call at the unchanged
  measure: \texttt{IHswap\_of\_IH} merely transports every already-available
  lower-goal hypothesis across the identity
  \(\mu_B(a,b,y)=\mu_B(b,a,y)\).  The two weak inequalities give
  \(ay\GameEq by\).

  In case \(C(x_1,x_2,y)\), the strict comparison
  \(x_1\GameLt x_2\) supplies a bridge: either some right option \(x_1^R\)
  satisfies \(x_1^R\GameLe x_2\), or some left option \(x_2^L\) satisfies
  \(x_1\GameLe x_2^L\).  Otherwise the recursive order definition would give
  \(x_2\GameLe x_1\), contradicting strictness.  In the first bridge case,
  a lower \(A\)-goal gives the adjacent cross inequality between \(x_1\) and
  \(x_1^R\).  Trichotomy compares \(x_1^R\) with \(x_2\): a strict comparison
  composes this adjacent inequality with a lower \(C\)-goal; equivalence uses
  a lower \(B\)-goal to replace the endpoint; the reverse strict case
  contradicts \(x_1^R\GameLe x_2\).  The bridge through \(x_2^L\) is
  symmetric.  The same construction is carried out for left and right options
  of \(y\), proving every required \(C_L\) and \(C_R\) inequality.

  In all three cases, every recursive invocation is accompanied in Lean by a
  specific goal-lowering theorem, so the simultaneous argument closes by
  well-founded induction rather than by mutual recursion assumed informally.
\end{proof}

\begin{theorem}[Conway A: closure of the product]
  \label{thm:conway-a}
  \lean{Conway.conway_A}
  \leanok
  If \(x\) and \(y\) are surreal games, then \(xy\) is surreal.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:abc-main}
  Apply \texttt{ABC\_main} to the constructor \texttt{Goal.A} and supply the
  two surrealness hypotheses.  The \texttt{A} branch of \texttt{Holds} is
  definitionally the assertion that \(xy\) is surreal.
\end{proof}

\begin{theorem}[Conway B: product respects equivalence]
  \label{thm:conway-b}
  \lean{Conway.conway_B}
  \leanok
  If \(x_1,x_2,y\) are surreal and \(x_1\GameEq x_2\), then
  \(x_1y\GameEq x_2y\).

  \leanok
  \uses{thm:abc-main}

  \texttt{Holds} yields \(x_1y\GameEq x_2y\).
\end{proof}

\section{Congruence and associativity}

\begin{theorem}[Two-variable multiplicative congruence]
  \label{thm:game-mul-congr}
  \lean{Game.mul_congr}
  \leanok
  If \(x_1\GameEq x_2\) and \(y_1\GameEq y_2\), and all four games are surreal, then
  \[
    x_1y_1\GameEq x_2y_2.
  \]
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:game-order-calculus,thm:game-mul-laws,thm:conway-b}
  Conway B applied directly to \(x_1\GameEq x_2\) gives
  \(x_1y_1\GameEq x_2y_1\).  To replace the second factor, commute both
  products, apply Conway B to \(y_1\GameEq y_2\) with common factor \(x_2\),
  and commute back, obtaining \(x_2y_1\GameEq x_2y_2\).  Transitivity of game
  equivalence composes these two replacements.
\end{proof}

\begin{theorem}[Associativity on surreal games]
  \label{thm:game-mul-assoc}
  \lean{Game.mul_assoc_of_isSurreal}
  \leanok
  If \(a,b,c\) are surreal games, then
  \[
    (ab)c\GameEq a(bc).
  \]
\end{theorem}
\begin{proof}
  \leanok
  \uses{def:game-inductions,thm:option-extensionality,thm:surreal-add-closure,thm:game-mul-laws,thm:game-distrib,thm:conway-a,thm:conway-b,thm:game-mul-congr}
  The proof uses well-founded induction on
  \(\bd(a)+\bd(b)+\bd(c)\), but packages a strengthened hypothesis:
  associativity may be invoked for any triple whose coordinate birthdays are
  bounded by those of \((a,b,c)\), provided at least one bound is strict.
  Thus all seven nonempty ways of replacing coordinates by options are
  available in a single step.

  A preparatory quotient calculation expands a product option under either
  bracketing.  It applies the seven smaller associativity equivalences, the
  game-level distributive laws, the compatibility of products with
  equivalence, and the corresponding formula for multiplication by a
  negative.  After passing to \(\mathcal G/{\GameEq}\), \texttt{abel}
  normalizes the resulting signed sum and proves the two expanded
  \(\mulopt\) expressions equivalent.

  We illustrate the option matching in two sign patterns.  First choose an
  \(LL\) option of \((ab)c\), and suppose its selected left option of \(ab\)
  is itself of type \(LL\).  For
  \(a^L\in A^L,b^L\in B^L,c^L\in C^L\), this option is
  \[
    L=\mulopt\bigl(\mulopt(a^L,b;a,b^L),c;ab,c^L\bigr).
  \]
  On the other bracketing, \(\mulopt(b^L,c;b,c^L)\) is an \(LL\) left option
  of \(bc\), so
  \[
    L'=\mulopt\bigl(a^L,bc;a,\mulopt(b^L,c;b,c^L)\bigr)
  \]
  is an \(LL\) left option of \(a(bc)\).  The preparatory seven-hypothesis
  rewrite gives \(L\GameEq L'\).

  As a mixed example, choose an \(LR\) right option of \((ab)c\), again using
  an \(LL\) left option of \(ab\).  With \(c^R\in C^R\), it is
  \[
    R=\mulopt\bigl(\mulopt(a^L,b;a,b^L),c;ab,c^R\bigr).
  \]
  Now \(\mulopt(b^L,c;b,c^R)\) is an \(LR\) right option of \(bc\), and hence
  \[
    R'=\mulopt\bigl(a^L,bc;a,\mulopt(b^L,c;b,c^R)\bigr)
  \]
  is an \(LR\) right option of \(a(bc)\).  The same rewrite, with
  \((a^L,b^L,c^R)\), proves \(R\GameEq R'\).

  The remaining branches are obtained by choosing \(a^R,b^R,c^L,c^R\) with
  the signs prescribed by the outer and inner option families.  The forward
  left- and right-option matching lemmas enumerate the four branches from
  \((ab)c\) to \(a(bc)\); the two symmetric matching lemmas enumerate the
  reverse direction.  All products required by the seven associativity hypotheses are
  surreal by Conway A, since options of surreal games are surreal.  Therefore
  the four matching lemmas satisfy option extensionality and yield
  \((ab)c\GameEq a(bc)\).
\end{proof}
